module

public import MathlibExt.Analysis.CStarAlgebra.CyclicOrbitProjection

/-!
# Original double-commutant operators preserve the actual closed cyclic orbit

The actual orbit projection commutes with the original source star algebra,
so it belongs to that algebra's commutant. An operator in the original
double commutant commutes with this genuine projection. Since the projection
fixes ξ, it fixes Tξ. The stock projection membership law places Tξ in the
actual norm-closed original orbit. No projection, commutation oracle,
completion, closure or representation model is supplied as an input.
-/

@[expose] public section

set_option autoImplicit false
set_option relaxedAutoImplicit false

namespace MathlibExt.Analysis.CStarAlgebra

noncomputable section

universe u

variable {H : Type u} [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H]

/-- An original double-commutant operator sends ξ into the actual closed cyclic orbit of the source algebra. -/
theorem bicommutant_apply_mem_cyclicOrbitSubspace
    (S : StarSubalgebra ℂ (H →L[ℂ] H)) (T : H →L[ℂ] H)
    (hT : T ∈ Set.centralizer (Set.centralizer (S : Set (H →L[ℂ] H)))) (ξ : H) :
    T ξ ∈ cyclicOrbitSubspace S ξ := by
  let P : H →L[ℂ] H := cyclicOrbitProjection S ξ
  have hcommutant : P ∈ Set.centralizer (S : Set (H →L[ℂ] H)) := by
    apply Set.mem_centralizer_iff.mpr
    intro a ha
    exact (cyclicOrbitProjection_commutes S ξ ⟨a, ha⟩).symm
  have hPT : P * T = T * P := (Set.mem_centralizer_iff.mp hT) P hcommutant
  have hPξ : P ξ = ξ := cyclicOrbitProjection_apply_self S ξ
  have hfix : P (T ξ) = T ξ := by
    calc
      P (T ξ) = (P * T) ξ := rfl
      _ = (T * P) ξ := congrArg (fun A : H →L[ℂ] H => A ξ) hPT
      _ = T ξ := by rw [mul_apply_eq_comp, hPξ]
  let K := cyclicOrbitSubspace S ξ
  let : CompleteSpace K := by
    change CompleteSpace
      (Submodule.span ℂ (Set.range fun a : S => (a : H →L[ℂ] H) ξ)).topologicalClosure
    infer_instance
  have hprojection : K.starProjection (T ξ) = T ξ := by
    change P (T ξ) = T ξ
    exact hfix
  exact Submodule.starProjection_eq_self_iff.mp hprojection

end

end MathlibExt.Analysis.CStarAlgebra
