public import Mathlib.Topology.Algebra.Module.Basic

/-!
# The actual closed cyclic orbit projection of an original star algebra

The norm closure of the complex span of the literal operator orbit Sξ is
an actual closed Hilbert subspace. Its stock orthogonal projection is a
genuine star projection fixing ξ. Multiplication and star closure of the
original unital star algebra make that subspace reducing, so the actual
projection commutes with every original source operator. No completed
space, reducing subspace, projection or commutation certificate is input.
-/

@[expose] public section

set_option autoImplicit false
set_option relaxedAutoImplicit false

open scoped InnerProductSpace

namespace MathlibExt.Analysis.CStarAlgebra

noncomputable section

universe u

variable {H : Type u} [NormedAddCommGroup H] [InnerProductSpace ℂ H] [CompleteSpace H]

/-- The actual norm-closed linear orbit of the original operator star algebra at ξ. -/
def cyclicOrbitSubspace (S : StarSubalgebra ℂ (H →L[ℂ] H)) (ξ : H) : Submodule ℂ H :=
  ContinuousLinearMap.orbitSubspace (S : Set (H →L[ℂ] H)) ξ

/-- The stock orthogonal projection onto that actual closed orbit subspace. -/
def cyclicOrbitProjection (S : StarSubalgebra ℂ (H →L[ℂ] H)) (ξ : H) : H →L[ℂ] H := by
  let K := cyclicOrbitSubspace S ξ
  let : CompleteSpace K := by
    change CompleteSpace
      (Submodule.span ℂ (Set.range fun a : S => (a : H →L[ℂ] H) ξ)).topologicalClosure
    infer_instance
  exact K.starProjection

/-- Original source-algebra multiplication preserves the actual closed cyclic orbit. -/
theorem cyclicOrbitSubspace_invariant (S : StarSubalgebra ℂ (H →L[ℂ] H)) (ξ : H)
    (a : S) (v : H) (hv : v ∈ cyclicOrbitSubspace S ξ) :
    (a : H →L[ℂ] H) v ∈ cyclicOrbitSubspace S ξ := by
  let R := Submodule.span ℂ (Set.range fun b : S => (b : H →L[ℂ] H) ξ)
  let K := cyclicOrbitSubspace S ξ
  let A : H →L[ℂ] H := a
  have hclosed : IsClosed (K : Set H) := Submodule.isClosed_topologicalClosure R
  have hspan : R ≤ K.comap A.toLinearMap := by
    apply Submodule.span_le.mpr
    rintro w ⟨b, rfl⟩
    change A ((b : H →L[ℂ] H) ξ) ∈ K
    have hgen : ((a * b : S) : H →L[ℂ] H) ξ ∈ R :=
      Submodule.subset_span (Set.mem_range_self (a * b))
    have hmem := Submodule.le_topologicalClosure R hgen
    change ((a : H →L[ℂ] H) * (b : H →L[ℂ] H)) ξ ∈ K at hmem
    rw [mul_apply_eq_comp] at hmem
    exact hmem
  have hpreimage : IsClosed (K.comap A.toLinearMap : Set H) := hclosed.preimage A.continuous
  exact Submodule.topologicalClosure_minimal R hspan hpreimage hv

/-- The actual closed-orbit projection is a genuine star projection. -/
theorem isStarProjection_cyclicOrbitProjection (S : StarSubalgebra ℂ (H →L[ℂ] H)) (ξ : H) :
    IsStarProjection (cyclicOrbitProjection S ξ) := by
  let K := cyclicOrbitSubspace S ξ
  let : CompleteSpace K := by
    change CompleteSpace
      (Submodule.span ℂ (Set.range fun a : S => (a : H →L[ℂ] H) ξ)).topologicalClosure
    infer_instance
  change IsStarProjection K.starProjection
  exact isStarProjection_starProjection

/-- Original unitality places ξ in its actual cyclic orbit, so its projection fixes ξ. -/
theorem cyclicOrbitProjection_apply_self (S : StarSubalgebra ℂ (H →L[ℂ] H)) (ξ : H) :
    cyclicOrbitProjection S ξ ξ = ξ := by
  let K := cyclicOrbitSubspace S ξ
  let : CompleteSpace K := by
    change CompleteSpace
      (Submodule.span ℂ (Set.range fun a : S => (a : H →L[ℂ] H) ξ)).topologicalClosure
    infer_instance
  change K.starProjection ξ = ξ
  apply Submodule.starProjection_eq_self_iff.mpr
  change ξ ∈ (Submodule.span ℂ (Set.range fun a : S => (a : H →L[ℂ] H) ξ)).topologicalClosure
  apply Submodule.le_topologicalClosure
  have hmem : (1 : H →L[ℂ] H) ξ ∈
      Submodule.span ℂ (Set.range fun a : S => (a : H →L[ℂ] H) ξ) :=
    Submodule.subset_span (Set.mem_range_self (1 : S))
  simpa only [one_apply_eq_self] using hmem

/-- The actual projection commutes with every original operator of the source star algebra. -/
theorem cyclicOrbitProjection_commutes (S : StarSubalgebra ℂ (H →L[ℂ] H)) (ξ : H) (a : S) :
    cyclicOrbitProjection S ξ * (a : H →L[ℂ] H) = (a : H →L[ℂ] H) * cyclicOrbitProjection S ξ := by
  let K := cyclicOrbitSubspace S ξ
  let : CompleteSpace K := by
    change CompleteSpace
      (Submodule.span ℂ (Set.range fun b : S => (b : H →L[ℂ] H) ξ)).topologicalClosure
    infer_instance
  let A : H →L[ℂ] H := a
  apply ContinuousLinearMap.ext
  intro v
  change K.starProjection (A v) = A (K.starProjection v)
  apply Submodule.eq_starProjection_of_mem_of_inner_eq_zero
  · exact cyclicOrbitSubspace_invariant S ξ a _ (Submodule.starProjection_apply_mem K v)
  · intro z hz
    rw [← A.map_sub, ← ContinuousLinearMap.adjoint_inner_right]
    have hstar : ContinuousLinearMap.adjoint A z ∈ K := by
      have h := cyclicOrbitSubspace_invariant S ξ (star a) z hz
      change (star A) z ∈ K at h
      simpa only [ContinuousLinearMap.star_eq_adjoint] using h
    exact (Submodule.mem_orthogonal' K _).mp (Submodule.sub_starProjection_mem_orthogonal (K := K) v) _ hstar

end

end MathlibExt.Analysis.CStarAlgebra
